SELECT r.id_remedio, r.nome, r.principio_ativo, COUNT(*) AS total_prescricoes FROM prescricao pr JOIN atendimento a ON a.id_atend = pr.id_atend JOIN remedio r     ON r.id_remedio = pr.id_remedio WHERE a.tipo = 'Emergencia' GROUP BY r.id_remedio, r.nome, r.principio_ativo ORDER BY total_prescricoes DESC, r.id_remedio LIMIT 10
